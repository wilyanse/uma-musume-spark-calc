#!/usr/bin/env bash
# T19 fresh gates. Writes one log per gate under $OUT. Scratch only.
set -u
W=/Users/willjose/Projects/personal/uma-musume-spark-calc.worktrees/T19-agent-recipient-notices
S=/Users/willjose/.hermes/cache/scratch/T19
OUT=$S/gates
export DOTNET_ROOT=/Users/willjose/.hermes/cache/scratch/T16b/dotnet-combo
export DOTNET_BIN=$DOTNET_ROOT/dotnet
export TMPDIR=$S/tmp
D=$DOTNET_BIN
rm -rf "$OUT" "$S/final-publish" "$S/tmp"; mkdir -p "$OUT" "$S/tmp"
cd "$W"
find agent -type d \( -name bin -o -name obj \) -prune -exec rm -rf {} +
run() { local name=$1; shift; echo "\$ $*" >"$OUT/$name.log"; "$@" >>"$OUT/$name.log" 2>&1; local rc=$?; echo "exit=$rc" >>"$OUT/$name.log"; echo "$name exit=$rc"; }
run 01-runtime bash .github/scripts/agent-gates.sh runtime
run 02-core-tests bash .github/scripts/agent-gates.sh core-tests
run 03-windows-build bash .github/scripts/agent-gates.sh windows-build
run 04-portable-tests bash -c "cd agent && $D test tests/Uma.Agent.Windows.Tests/Portable/Uma.Agent.Windows.Portable.Tests.csproj -c Release"
run 05-native-build bash -c "cd agent && $D build tests/Uma.Agent.Windows.Tests/Native/Uma.Agent.Windows.Tests.csproj -c Release -p:EnableWindowsTargeting=true"
run 06-solution-build bash -c "cd agent && $D build Uma.Agent.sln -c Release -p:EnableWindowsTargeting=true"
run 07-publish bash -c "cd agent && $D publish src/Uma.Agent.Windows/Uma.Agent.Windows.csproj -c Release -r win-x64 --self-contained true -p:PublishSingleFile=true -p:IncludeNativeLibrariesForSelfExtract=true -p:EnableWindowsTargeting=true -o $S/final-publish && ls -l $S/final-publish"
run 08-notice-regressions python3 -X dev -W error agent/legal/tools/test_recipient_notices.py
run 09-check-legal python3 agent/legal/tools/recipient_notices.py check-legal
run 10-verify-exe python3 agent/legal/tools/recipient_notices.py verify-exe "$S/final-publish/Uma.Agent.Windows.exe" --assets agent/src/Uma.Agent.Windows/obj/project.assets.json
rm -f "$S/final.zip"
run 11-package-zip python3 agent/legal/tools/recipient_notices.py package "$S/final-publish/Uma.Agent.Windows.exe" "$S/final.zip" --assets agent/src/Uma.Agent.Windows/obj/project.assets.json
run 12-verify-zip python3 agent/legal/tools/recipient_notices.py verify-zip "$S/final.zip" --assets agent/src/Uma.Agent.Windows/obj/project.assets.json
run 13-zip-list unzip -l "$S/final.zip"
run 14-bundle-data python3 agent/tests/Uma.Agent.Windows.Tests/Tools/bundle_probe.py "$S/final-publish/Uma.Agent.Windows.exe" "$W"
run 15-native-sqlite python3 agent/tests/Uma.Agent.Windows.Tests/Tools/sqlite_native_probe.py "$S/final-publish/Uma.Agent.Windows.exe" /Users/willjose/.hermes/cache/scratch/T18/sqlitepclraw.lib.e_sqlite3.2.1.12.nupkg
for p in src/Uma.Agent.Core/Uma.Agent.Core.csproj src/Uma.Agent.Windows/Uma.Agent.Windows.csproj tests/Uma.Agent.Core.Tests/Uma.Agent.Core.Tests.csproj tests/Uma.Agent.Windows.Tests/Portable/Uma.Agent.Windows.Portable.Tests.csproj tests/Uma.Agent.Windows.Tests/Native/Uma.Agent.Windows.Tests.csproj; do
  n=$(basename "$p" .csproj)
  run "16-audit-$n" bash -c "cd agent && $D list $p package --vulnerable --include-transitive"
done
run 17-audit-props bash -c "cd agent && $D msbuild src/Uma.Agent.Windows/Uma.Agent.Windows.csproj -getProperty:NuGetAudit -getProperty:NuGetAuditMode -getProperty:NuGetAuditLevel -getProperty:TreatWarningsAsErrors -p:EnableWindowsTargeting=true"
run 18-github-regressions env TOOLS_DIR=/Users/willjose/.hermes/cache/scratch/IR1/tools bash .github/tests/run.sh
run 19-check-config node .github/scripts/check-config.mjs .
run 20-shipped-graph python3 agent/legal/tools/recipient_notices.py report "$S/final-publish/Uma.Agent.Windows.exe"
