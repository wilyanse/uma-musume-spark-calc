"""Sol fix2 actual network/archive observations; no inventory value copied as observation."""
from pathlib import Path
import hashlib
import urllib.request
import zipfile
import xml.etree.ElementTree as ET
import json
S = Path(__file__).parent
W = Path('/Users/willjose/Projects/personal/uma-musume-spark-calc.worktrees/T19-agent-recipient-notices')
inv = json.loads((W / 'agent/legal/inventory.json').read_bytes())
records = [c for c in inv['components'] if c['deps'].startswith('runtimepack.Microsoft.NETCore.App.Runtime.')]
records += [dict(inv['singleFileHost'], deps=inv['singleFileHost']['package'] + '/' + inv['singleFileHost']['version'])]
for r in records:
    name, version = r['deps'].removeprefix('runtimepack.').split('/')
    file = name.lower() + '.' + version + '.nupkg'
    url = 'https://api.nuget.org/v3-flatcontainer/' + name.lower() + '/' + version + '/' + file
    print('GET', url, flush=True)
    with urllib.request.urlopen(url, timeout=120) as response:
        b = response.read(); print('HTTP', response.status)
    (S / file).write_bytes(b)
    digest = hashlib.sha256(b).hexdigest()
    print('sha256', digest, 'inventory hash matches', digest == r['nupkgSha256'])
    assert digest == r['nupkgSha256']
    with zipfile.ZipFile(S / file) as z:
        member = next(n for n in z.namelist() if n.endswith('.nuspec'))
        xml = z.read(member); (S / (file + '.nuspec')).write_bytes(xml)
        repo = next(e.attrib for e in ET.fromstring(xml).iter() if e.tag.split('}')[-1] == 'repository')
        print('archive nuspec member', member, 'repository', repo)
        for text in ('LICENSE.TXT', 'THIRD-PARTY-NOTICES.TXT'):
            archive = z.read(text)
            delivered = (W / 'agent/legal/notices' / r['packTexts'][text]).read_bytes()
            assert archive == delivered
            url = 'https://raw.githubusercontent.com/dotnet/runtime/' + repo['commit'] + '/' + text
            print('GET', url, flush=True)
            with urllib.request.urlopen(url, timeout=120) as response:
                upstream = response.read(); print('HTTP', response.status)
            (S / (name.lower() + '-' + text)).write_bytes(upstream)
            same = archive.replace(b'\r\n', b'\n') == upstream.replace(b'\r\n', b'\n')
            print('archive/delivered exact match True; upstream CRLF/LF normalized match', same, 'upstream sha256', hashlib.sha256(upstream).hexdigest())
            assert same
print('Fresh download/source observations verified; no package pins/hashes/licence bytes changed')
