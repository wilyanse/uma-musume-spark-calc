failures = 0
for name, args, cwd, timeout in commands:
    log = S / 'gates' / (name + '.log')
    if log.exists():
        previous = log.read_text()
        if previous.rstrip().endswith('exit=0'):
            print(name, 'already completed, not repeated', flush=True)
            continue
        saved = log.with_suffix('.interrupted.log')
        assert not saved.exists(), 'do not overwrite prior interrupted evidence'
        log.rename(saved)
    with log.open('w') as output:

        output.write('$ (cd %s; %s)\n' % (shlex.quote(str(cwd)), shlex.join(args)))
        output.flush()

            p = subprocess.run(args, cwd=cwd, env=env, stdout=output, stderr=subprocess.STDOUT, timeout=timeout)
