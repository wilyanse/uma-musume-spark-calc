def run(args):
    print('$ ' + shlex.join(args), flush=True)
    p = subprocess.run(args, cwd=W, env=env, capture_output=True, text=True, timeout=120)
    print(p.stdout + p.stderr, end='')
    print('exit=' + str(p.returncode), flush=True)
    assert p.returncode == 0
    return p.stdout


print('Sol fix2 committed-source/artifact verification, session20261011_103329_5bce0b')

assert run(['git', 'branch', '--show-current']).strip() == 'task/T19-agent-recipient-notices'

run(['git', '-c', 'core.whitespace=-blank-at-eol,-blank-at-eof,cr-at-eol', 'diff', '--check', base + '..HEAD'])
